% 附属注：编号为“上一注号+字母”，不占用下一个整数（如注1后接注1a，再下一条仍为注2）
\newenvironment{remarka}{%
  \addtocounter{remark}{-1}%
  \renewcommand{\theremark}{\arabic{remark}a}%
  \remark
}{%
  \endremark
}
